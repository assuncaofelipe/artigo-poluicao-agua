\begin{frame}{Coleta em Campo (Lago Serpa) e Aumento Controlado}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\linewidth}
      \centering
      \includegraphics[width=0.98\linewidth,height=4.5cm,keepaspectratio]{map_points_osm_21S.png}\\[2pt]
      {\scriptsize\color{gray}60 pontos georreferenciados no Lago Serpa (UTM 21S, Itacoatiara-AM).}
    \end{column}

    \begin{column}{0.48\linewidth}
      \textbf{Campanha Experimental:}
      \begin{itemize}\small
        \item \textbf{Amostras Reais:} 60 pontos coletados \textit{in situ} com sonda multiparâmetros.
        \item \textbf{Entradas (Inputs):} Temperatura, Condutividade Elétrica, TDS, Salinidade, Latitude e Longitude.
        \item \textbf{Alvos (Targets):} pH, Oxigênio Dissolvido (OD) e Turbidez (TURB).
      \end{itemize}
      \vspace{0.15cm}
      \begin{block}{Aumento Gaussiano Controlado ($\times 10$)}
        \footnotesize
        $\bullet$ Expansão do suporte de treino de 60 para 600 amostras ($\sigma = 0{,}01$).\\
        $\bullet$ Redução de 2,53\% no MAE macro e estabilização de gradientes em regime de \textit{small-data}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% SLIDE 6: Arquitetura Multimodal SNN Ponta a Ponta (1:05)
% -------------------------------------------------------------
\begin{frame}{Arquitetura da Rede Neural de Spikes (SNN)}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\linewidth}
      \centering
      \includegraphics[width=0.99\linewidth,height=4.5cm,keepaspectratio]{Arquitetura.png}\\[2pt]
      {\scriptsize\color{gray}Fluxo: Entradas $\to$ Codificação em Spikes $\to$ CUBA-LIF $\to$ Decodificação Multimodal.}
    \end{column}

    \begin{column}{0.46\linewidth}
      \textbf{Parâmetros da Rede Neuromórfica:}
      \begin{itemize}\small
        \item \textbf{Camada de Entrada:} 6 variáveis normalizadas em $[0, 1]$.
        \item \textbf{Codificação Temporal:} Passos de tempo $T = 16\text{--}32$ passos.
        \item \textbf{Neurônios Ocultos:} Dinâmica Leaky Integrate-and-Fire (LIF) com gradiente substituto (\textit{surrogate gradient}).
        \item \textbf{Saída Multimodal:} 3 saídas lineares integradas no tempo para regressão contínua.
      \end{itemize}
      \vspace{0.15cm}
      \begin{exampleblock}{Otimização por Grid Search}
        \scriptsize
        Varredura sistemática de taxa de aprendizado, limiar ($\theta$), decaimento de membrana ($\beta$) e pesos balanceados da perda.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

          ESP32 (Xtensa Dual-Core 240 MHz, MicroPython / C++).
          \vspace{0.06cm}
          \item \textbf{Computador em Placa (SBC):}\\
          Raspberry Pi 4B (Quad-Core Cortex-A72 @ 1.5 GHz).
          \vspace{0.06cm}
          \item \textbf{Smartphones Android:}\\
          Três aparelhos heterogêneos (Entrada, Intermediário e Alta Eficiência).
        \end{itemize}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.47\linewidth}
      \begin{alertblock}{Setup de Medição Energética Física}
        \footnotesize
        \begin{itemize}
          \item \textbf{Sensor de Corrente INA219:}\\
          Medição em shunt no trilho de alimentação do ESP32 e RPi 4B.
          \vspace{0.06cm}
          \item \textbf{Potência Incremental:}\\
          $\Delta P = P_{\text{load}} - P_{\text{idle}}$ (repouso vs inferência).
          \vspace{0.06cm}
          \item \textbf{Energia por Inferência:}\\
          $E = \frac{\Delta P}{\text{Throughput}}$ (em mJ/inf).
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Resultados e Conclusões}

% SLIDE 8: Desempenho Preditivo e Significância Estatística (1:15)
% -------------------------------------------------------------
\begin{frame}{Desempenho Preditivo Multimodal ($n=10$ Seeds)}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\linewidth}
      \centering
      \includegraphics[width=0.99\linewidth,height=4.6cm,keepaspectratio]{mae_vs_model_bar.png}\\[2pt]
      {\scriptsize\color{gray}Comparativo de MAE macro entre modelos (IC 95\%).}
    \end{column}

    \begin{column}{0.46\linewidth}
      \textbf{Métricas Macro ($n=10$ seeds):}
      \begin{itemize}\small
        \item \textbf{Random Forest:} $\text{MAE} = 0{,}409 \pm 0{,}062$
        \item \textbf{LSTM:} $\text{MAE} = 0{,}428 \pm 0{,}059$
        \item \textbf{SNN (Proposta):} $\mathbf{\text{MAE} = 0{,}447 \pm 0{,}064}$
        \item \textbf{MLP Clássica:} $\text{MAE} = 0{,}781 \pm 0{,}193$ (instável)
      \end{itemize}
      \vspace{0.15cm}
      \begin{block}{Validação Estatística Não-Paramétrica}
        \scriptsize
        $\bullet$ \textbf{Wilcoxon Pareado:} Sem diferença significativa entre SNN e Random Forest ($p=0{,}064$) e SNN e LSTM ($p=0{,}105$).\\
        $\bullet$ \textbf{SNN vs MLP:} SNN supera MLP com $p=0{,}002$.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusões e Trabalhos Futuros}

      \begin{block}{Conclusões Centrais}
        \footnotesize
        \begin{enumerate}
          \item \textbf{Viabilidade Neuromórfica:} SNN multimodal competitiva com Random Forest e LSTM sem diferença estatística ($p>0{,}05$).
          \item \textbf{Operação 100\% Offline:} Validação prática em ESP32 ($11{,}2\text{ mJ}$), RPi 4B ($0{,}05\text{ mJ}$) e Android ($0{,}0028\text{ mJ}$).
          \item \textbf{Mapeamento Espacial:} Geração de superfícies contínuas IDW e Índice de Poluição para suporte à decisão.
          \item \textbf{Estabilidade com Poucos Dados:} Aumento gaussiano $\times 10$ reduz erro e estabiliza treino.
        \end{enumerate}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.47\linewidth}

          \item \textbf{Campanhas Sazonais:} Coleta de dados no período de cheia e seca dos rios amazônicos.
          \item \textbf{Boias Autônomas:} Integração com painéis solares para telemetria contínua com protocolo LoRaWAN.

% -------------------------------------------------------------
\begin{frame}{Backup 4: Formulação Matemática do Neurônio LIF}
  \footnotesize
  \begin{block}{Equações de Membrana em Tempo Discreto}
    \begin{equation*}
      U_i[t] = \beta U_i[t-1] + (1 - \beta) I_i[t] - S_i[t-1] \theta
    \end{equation*}
    \begin{equation*}
      S_i[t] = \Theta(U_i[t] - \theta), \quad \text{onde } \Theta(x) = \begin{cases} 1, & x \ge 0 \\ 0, & x < 0 \end{cases}
    \end{equation*}
  \end{block}
  \begin{exampleblock}{Gradiente Substituto (Surrogate Gradient - Fast Sigmoid)}
    \begin{equation*}
      \sigma'(x) = \frac{1}{(1 + k |x|)^2}, \quad k = 25
    \end{equation*}
    Permite a retropropagação através do tempo (BPTT) contornando a não-diferenciabilidade do disparo.
  \end{exampleblock}
\end{frame}

        \scriptsize

        \begin{equation*}
          I_p = w_1 \cdot \text{dev}(\text{pH}) + w_2 \cdot (1 - \text{norm}(\text{OD})) + w_3 \cdot \text{norm}(\text{TURB})
        \end{equation*}
        Valores próximos a $1{,}0$ indicam zonas de maior degradação hídrica.
